cmake_minimum_required(VERSION 3.24)
project(Veylo VERSION 0.5.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(RN "${CMAKE_SOURCE_DIR}/third_party/rnnoise")
set(RN_SOURCES denoise.c rnn.c pitch.c kiss_fft.c celt_lpc.c nnet.c nnet_default.c parse_lpcnet_weights.c rnnoise_data.c rnnoise_tables.c)
list(TRANSFORM RN_SOURCES PREPEND "${RN}/src/")
add_library(rnnoise STATIC ${RN_SOURCES} ${RN}/src/x86/x86cpu.c ${RN}/src/x86/x86_dnn_map.c ${RN}/src/x86/nnet_sse4_1.c ${RN}/src/x86/nnet_avx2.c)
target_include_directories(rnnoise PUBLIC "${RN}/include" PRIVATE "${RN}/src")
target_compile_definitions(rnnoise PRIVATE RNNOISE_BUILD RNN_ENABLE_X86_RTCD CPU_INFO_BY_C)
if(MSVC)
  set_source_files_properties(${RN}/src/x86/nnet_avx2.c PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
else()
  set_source_files_properties(${RN}/src/x86/nnet_sse4_1.c PROPERTIES COMPILE_OPTIONS "-msse4.1")
  set_source_files_properties(${RN}/src/x86/nnet_avx2.c PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma")
  target_compile_options(rnnoise PRIVATE -O3 -Wno-unknown-pragmas)
endif()
add_library(ses_native SHARED native/src/ses.cpp)
target_compile_definitions(ses_native PRIVATE SES_BUILD WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(ses_native PUBLIC native/include PRIVATE third_party native/src)
target_link_libraries(ses_native PRIVATE rnnoise ole32 uuid)
set_target_properties(ses_native PROPERTIES PREFIX "")
if(MSVC)
  target_compile_options(ses_native PRIVATE /W4 /permissive-)
else()
  target_compile_options(ses_native PRIVATE -Wall -Wextra -Wpedantic -mguard=cf)
  target_link_options(ses_native PRIVATE -static-libstdc++ -static-libgcc -mguard=cf)
endif()
enable_testing()
add_executable(ses_dsp_tests native/tests/dsp_tests.cpp)
target_link_libraries(ses_dsp_tests PRIVATE ses_native)
if(NOT MSVC)
  target_link_options(ses_dsp_tests PRIVATE -static-libstdc++ -static-libgcc)
endif()
add_test(NAME dsp COMMAND ses_dsp_tests)
add_executable(ses_resampler_tests native/tests/resampler_tests.cpp)
target_include_directories(ses_resampler_tests PRIVATE third_party)
if(NOT MSVC)
  target_link_options(ses_resampler_tests PRIVATE -static-libstdc++ -static-libgcc)
endif()
add_test(NAME resampler COMMAND ses_resampler_tests)
add_executable(ses_transport_tests native/tests/transport_tests.cpp)
if(NOT MSVC)
  target_link_options(ses_transport_tests PRIVATE -static-libstdc++ -static-libgcc)
endif()
add_test(NAME transport COMMAND ses_transport_tests)
add_executable(ses_fifo_tests native/tests/fifo_tests.cpp)
target_link_libraries(ses_fifo_tests PRIVATE ses_native)
if(NOT MSVC)
  target_link_options(ses_fifo_tests PRIVATE -static-libstdc++ -static-libgcc)
endif()
add_test(NAME fifo COMMAND ses_fifo_tests)
add_executable(ses_driver_lab_tests native/tests/driver_lab_tests.cpp)
if(NOT MSVC)
  target_link_options(ses_driver_lab_tests PRIVATE -static-libstdc++ -static-libgcc)
endif()
# Deliberately not a CTest: active malformed kernel requests belong only in the lab.
option(SES_SANITIZE "AddressSanitizer checks for development builds" OFF)
if(SES_SANITIZE AND NOT MSVC)
  foreach(target rnnoise ses_native ses_dsp_tests ses_resampler_tests ses_transport_tests ses_fifo_tests)
    target_compile_options(${target} PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address)
  endforeach()
endif()
